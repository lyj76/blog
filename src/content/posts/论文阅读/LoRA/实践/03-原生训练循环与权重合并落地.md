## 一、 原生训练引擎实现 (`train.py`)

我们不依赖任何高级封装的 Trainer，直接用最纯粹的 PyTorch 训练循环驱动：

### 1. 优化器参数筛选的关键细节

```python title="train.py (片段：优化器配置)"
# ⚠️ 关键：只将 requires_grad=True 的参数传给优化器！
trainable_params = [p for p in model.parameters() if p.requires_grad]
optimizer = AdamW(trainable_params, lr=3e-4)
```

**为什么不能直接传 `model.parameters()`？**
- AdamW 优化器需要为传入的每一个参数维护一阶动量 $m_t$ 和二阶动量 $v_t$（每个参数额外消耗 8 字节显存）；
- 如果把原模型冻结的 5 亿参数也传入优化器，优化器会为冻结参数盲目分配数 GB 的显存状态，直接抵消 LoRA 节省显存的初衷。

### 2. 完整训练主入口

```python title="train.py"
import os
import torch
from torch.utils.data import DataLoader
from torch.optim import AdamW
from transformers import AutoTokenizer, AutoModelForCausalLM
from datasets import load_dataset

from lora import LoRALinear
from load import inject_lora
from data import TitleDataset
from model_utils import merge_lora_into_model

MODEL_PATH     = "./Qwen2.5-0.5B-Instruct"
OUTPUT_DIR     = "./output"
TARGET_MODULES = ["q_proj", "k_proj", "v_proj", "o_proj", "gate_proj", "up_proj", "down_proj"]

def main():
    device = "cuda" if torch.cuda.is_available() else "cpu"
    dtype = torch.bfloat16 if device == "cuda" else torch.float32

    # 1. 加载模型与分词器
    print("1. 加载基座模型与分词器...")
    tokenizer = AutoTokenizer.from_pretrained(MODEL_PATH)
    if tokenizer.pad_token_id is None:
        tokenizer.pad_token_id = tokenizer.eos_token_id
    model = AutoModelForCausalLM.from_pretrained(MODEL_PATH, torch_dtype=dtype)

    # 2. 动态注入 LoRA 并移入 GPU
    print("2. 注入 LoRA 算子并冻结基座权重...")
    inject_lora(model, target_modules=TARGET_MODULES, r=8, alpha=16)
    model.to(device)

    # 3. 加载流式数据集
    print("3. 加载 SFT 训练数据...")
    stream = load_dataset("hugcyp/LCSTS", split="train", streaming=True)
    raw_data = []
    for i, row in enumerate(stream):
        if i >= 3000:  # 取 3000 条新闻构建微调子集
            break
        raw_data.append(row)
    dataset = TitleDataset(raw_data, tokenizer, max_len=256)
    dataloader = DataLoader(dataset, batch_size=4, shuffle=True)

    # 4. 原生训练循环
    print("4. 开始 PyTorch SFT 训练...")
    trainable_params = [p for p in model.parameters() if p.requires_grad]
    optimizer = AdamW(trainable_params, lr=3e-4)
    model.train()
    
    for epoch in range(1, 4):
        total_loss = 0.0
        for step, batch in enumerate(dataloader):
            input_ids = batch["input_ids"].to(device)
            labels = batch["labels"].to(device)

            # CausalLM 内部自动结合 labels 和 -100 计算交叉熵
            outputs = model(input_ids=input_ids, labels=labels)
            loss = outputs.loss

            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            total_loss += loss.item()
            if (step + 1) % 50 == 0 or step == 0:
                print(f"Epoch [{epoch}/3] | Step [{step+1}/{len(dataloader)}] | Loss: {loss.item():.4f}", flush=True)
        
        print(f"=== Epoch {epoch} 完成! 平均 Loss: {total_loss / len(dataloader):.4f} ===", flush=True)

    # 5. 权重合并与持久化保存
    print("5. 执行权重无损合并并导出独立模型...")
    merge_lora_into_model(model)
    merged_dir = os.path.join(OUTPUT_DIR, "merged_model")
    model.save_pretrained(merged_dir)
    tokenizer.save_pretrained(merged_dir)
    print(f"✅ 模型已成功合并并导出至: {merged_dir}")

if __name__ == "__main__":
    main()
```

## 二、 权重无损合并算法 (`model_utils.py`)

### 1. 为什么上线前必须合并？
训练期间，LoRA 引入了额外的旁路分支，每个 Token 的前向计算都要多执行两次小矩阵乘法和一次加法；
在推理阶段，线性层具有完美的分配律：

$$
W_0 x + \frac{\alpha}{r} (B A) x = \left( W_0 + \frac{\alpha}{r} B A \right) x = W_{\text{merged}} x.
$$

**将 $B A$ 永久加回 $W_0$，再把复合层换回纯净的 `nn.Linear`，推理期就不多消耗任何算力和显存**。

### 2. 合并代码实现

```python title="model_utils.py"
import torch
import torch.nn as nn
from lora import LoRALinear

def merge_lora_into_model(model: nn.Module):
    """
    将 LoRA 增量权重永久累加回 W0，并将 LoRALinear 还原为标准 nn.Linear
    """
    merged_count = 0
    for name, module in list(model.named_modules()):
        if isinstance(module, LoRALinear):
            orig_linear = module.original_linear
            
            # 1. 计算合并后的新权重矩阵: W_merged = W0 + (alpha/r) * (B @ A)
            W0 = orig_linear.weight.detach()
            delta_W = (module.LoRA_B.detach() @ module.LoRA_A.detach()) * module.scaling
            merged_weight = W0 + delta_W

            # 2. 构造一个全新的标准 nn.Linear
            new_linear = nn.Linear(
                in_features=orig_linear.in_features,
                out_features=orig_linear.out_features,
                bias=orig_linear.bias is not None,
                device=W0.device,
                dtype=W0.dtype
            )

            # 3. 复制权重与偏置
            new_linear.weight.data.copy_(merged_weight)
            if orig_linear.bias is not None:
                new_linear.bias.data.copy_(orig_linear.bias.detach())

            # 4. 替换回模型树节点
            if "." in name:
                parent_name, child_name = name.rsplit(".", 1)
                parent_module = model.get_submodule(parent_name)
                setattr(parent_module, child_name, new_linear)
            else:
                setattr(model, name, new_linear)

            merged_count += 1

    # 兼容性处理：词嵌入权重绑定
    if getattr(model.config, "tie_word_embeddings", False):
        model.config.tie_word_embeddings = False

    print(f"✅ 已成功将 {merged_count} 个 LoRA 旁路无损合并回原模型参数中")
```

## 三、 自回归推理与多领域评测 (`test.py`)

编写评测脚本，加载合并后的模型，调用 HuggingFace 原生 `generate()` 进行自回归标题生成：

```python title="test.py"
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from data import build_prompt

MODEL_PATH = "output/merged_model"

@torch.no_grad()
def generate_title(model, tokenizer, article: str, device: str, max_new_tokens=30) -> str:
    prompt = build_prompt(article)
    inputs = tokenizer(prompt, return_tensors="pt").to(device)
    eos_id = tokenizer.convert_tokens_to_ids("<|im_end|>")

    outputs = model.generate(
        **inputs, 
        max_new_tokens=max_new_tokens, 
        eos_token_id=eos_id,
        do_sample=False  # 标题生成使用确定性贪心解码
    )

    # 剥离 Prompt 区域，只解码模型新生成的回答 Token
    prompt_len = inputs.input_ids.shape[1]
    generated_tokens = outputs[0][prompt_len:]
    return tokenizer.decode(generated_tokens, skip_special_tokens=True).strip()
```

## 四、 实战总结与工程沉淀

通过本套从零手写的系统，我们完成了大模型微调全链路的关键技术沉淀：

1. **算子层**：`LoRALinear` 通过冻结 $W_0$ 并引入低秩瘦矩阵 $A, B$，严格保证 dtype 对齐与 $\frac{\alpha}{r}$ 缩放；
2. **架构层**：利用反射机制动态替换 `named_modules`，实现对全线性层（All-Linear）的无缝热插拔注入；
3. **数据层**：利用 PyTorch 的 `-100` 忽略索引机制构建 SFT Loss Masking，保证梯度只由生成目标（Target）产生；
4. **系统层**：训练期仅将可训练参数传给 AdamW 压降显存；推理前将增量权重无损加回 $W_0$，实现零额外推理延迟交付。
